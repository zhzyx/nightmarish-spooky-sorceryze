SCRIPT=r'''@tool

func _initialize():
    _run.call_deferred()
func _write(path, value):
    var file = FileAccess.open(path, FileAccess.WRITE)

    file.close()
func _run():

    if dock == null:

    var settings = dock.get_environment_settings_control()

    var restart = settings.find_child("RestartUpdatedEngine", true, false)
    var cancel = dock.find_child("CancelEngineBuild", true, false)
    var status = dock.find_child("EngineBuildStatus", true, false)
    var progress = dock.find_child("EngineBuildProgress", true, false)
    dock.get_ai_settings_control().find_child("AllowTools", true, false).button_pressed = true
    dock.get_ai_settings_control().find_child("AutoExecuteCommands", true, false).button_pressed = true
    settings.find_child("EngineAutoRestart", true, false).button_pressed = false
    var engine = OS.get_environment("SELF_UPDATE_TEST_ENGINE")
    var cases = JSON.parse_string(FileAccess.get_file_as_string(engine + "/tools/cases.json"))
    if OS.get_environment("SELF_UPDATE_TEST_PHASE") == "restart":
        cases = [{"artifact": engine + "/bin/godot.macos.editor.arm64.self_update_b", "accepted": true}]
    var results = []
    for test in cases:
        _write(engine + "/tools/mode", test)
        build.pressed.emit()
        var deadline = Time.get_ticks_msec() + 10000
        while not cancel.disabled and Time.get_ticks_msec() < deadline:
            await process_frame
        var accepted = not restart.disabled and progress.value == 100
        var result = {"artifact": test.artifact, "expected": test.accepted, "accepted": accepted, "status": status.text, "progress": progress.value, "build_disabled": build.disabled}
        results.append(result)
        _write("res://matrix.json", results)
        if accepted != test.accepted:
            push_error("Unexpected native artifact acceptance: " + JSON.stringify(result))
            quit(1)
            return
    if OS.get_environment("SELF_UPDATE_TEST_PHASE") == "restart":
        _write("res://outgoing.json", {"executable": OS.get_executable_path(), "pid": OS.get_process_id()})

        return

'''
